    private static void Fresh()
    {

        CombatManager.Instance.History.Clear();

        var run = RunState.CreateForTest([owner, ally], seed: "OTTO-ANCIENT");
        state = new CombatState(runState: run);
        state.AddPlayer(owner); state.AddPlayer(ally); owner.ResetCombatState(); ally.ResetCombatState();
        enemy = state.CreateCreature(ModelDb.Monster<MockAttackMonster>().ToMutable(), CombatSide.Enemy, "enemy1"); state.AddCreature(enemy);

        typeof(CombatManager).GetField("_state", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(CombatManager.Instance, state);
        typeof(CombatManager).GetProperty("IsInProgress")!.SetValue(CombatManager.Instance, true);

    private static T New<T>(bool upgrade = false, PileType pile = PileType.Hand) where T : CardModel
    {
        var card = state.CreateCard<T>(owner);
        if (upgrade) card.UpgradeInternal();
        pile.GetPile(owner).AddInternal(card);
        return card;
    }
    private static async Task Play(CardModel card, Creature? target = null, bool auto = false)
    {
        // 每次从正确牌堆进入正式出牌包装器，覆盖易拉罐取位置、前后钩子、牌堆及重复出牌清理。
        if (!auto && card.Pile?.Type != PileType.Hand)
        {
            card.RemoveFromCurrentPile(); owner.PlayerCombatState!.Hand.AddInternal(card);
        }
        await card.OnPlayWrapper(Ctx, target, auto, default, skipCardPileVisuals: true);
    }
    private static bool Allowed(CardModel card) => Hook.ShouldPlay(state, card, out _, AutoPlayType.None);
    private static async Task StartTurn() => await Hook.AfterSideTurnStart(state, CombatSide.Player, [owner.Creature]);
    // 测试只派发状态的回合末清理，不启动真实敌方AI。
    private static async Task EndTurn()
    {
        foreach (var power in owner.Creature.Powers.ToArray())
            await power.AfterSideTurnEnd(Ctx, CombatSide.Player, [owner.Creature]);
    }
    private static void DrawPile(int count)
    {
        for (int i = 0; i < count; i++) New<OttoStrike>(pile: PileType.Draw);
    }
    private static async Task ClearPowers(Creature c)
    {
        foreach (var p in c.Powers.ToArray()) await PowerCmd.Remove(p);

    private static void SetMove(Creature target, params AbstractIntent[] intents)

        // 测试专用：改变内存目标的可见招式，不调用敌人的行动或AI。
        typeof(MonsterModel).GetProperty("NextMove")!.SetValue(target.Monster, new MoveState("PROBE", _ => Task.CompletedTask, intents));
    }

    private static void CheckHandSelectionReturn()
    {
        // 使用真实 Godot 容器和原版归还选中牌的方法；不构造完整战斗画面或修改任何存档。
        Fresh(); LocalContext.NetId=owner.NetId;
        var a=New<OttoStrike>();var chosen=New<ChopAgain>();var b=New<OttoDefend>();
        var source=New<ChihayaAnon>(pile:PileType.Play);
        var view=new MegaCrit.Sts2.Core.Nodes.Combat.NPlayerHand();testHand=view;
        var handContainer=new Control();view.AddChild(handContainer);
        typeof(MegaCrit.Sts2.Core.Nodes.Combat.NPlayerHand).GetProperty("CardHolderContainer")!.SetValue(view,handContainer);
        var selectedContainer=new MegaCrit.Sts2.Core.Nodes.Combat.NSelectedHandCardContainer();view.AddChild(selectedContainer);
        HarmonyLib.AccessTools.Field(view.GetType(),"_selectedHandCardContainer").SetValue(view,selectedContainer);
        var preview=new MegaCrit.Sts2.Core.Nodes.Cards.NUpgradePreview();view.AddChild(preview);
        HarmonyLib.AccessTools.Field(view.GetType(),"_upgradePreview").SetValue(view,preview);
        foreach(var model in new CardModel[]{a,chosen,b})
        {
            var node=new MegaCrit.Sts2.Core.Nodes.Cards.NCard();
            HarmonyLib.AccessTools.Field(node.GetType(),"_model").SetValue(node,model);
            MegaCrit.Sts2.Core.Nodes.Cards.Holders.NCardHolder holder=model==chosen
                ? new MegaCrit.Sts2.Core.Nodes.Cards.Holders.NSelectedHandCardHolder()
                : new MegaCrit.Sts2.Core.Nodes.Cards.Holders.NHandCardHolder();
            typeof(MegaCrit.Sts2.Core.Nodes.Cards.Holders.NCardHolder).GetProperty("CardNode")!.SetValue(holder,node);
            holder.AddChild(node);(model==chosen?(Node)selectedContainer:handContainer).AddChild(holder);
        }
        var patch=new HarmonyLib.Harmony("OTTOConfirmedProbe.HandFixture");
        patch.Patch(HarmonyLib.AccessTools.PropertyGetter(view.GetType(),"Instance"),
            prefix:new HarmonyLib.HarmonyMethod(typeof(Probe),nameof(UseTestHand)));
        try
        {
            var nativeReturn=HarmonyLib.AccessTools.Method(view.GetType(),"OnSelectModeSourceFinished").CreateDelegate<Action<AbstractModel>>(view);
            source.ExecutionFinished+=nativeReturn;
            OttoHandOrder.MoveLeft(chosen,source);
            Verify(owner.PlayerCombatState!.Hand.Cards.SequenceEqual(new CardModel[]{chosen,a,b}),"hand model moves immediately");
            Verify(view.GetCardHolder(chosen)!.GetParent()==selectedContainer,"selected visual remains until native return");
            source.InvokeExecutionFinished();
            var visible=handContainer.GetChildren().OfType<MegaCrit.Sts2.Core.Nodes.Cards.Holders.NHandCardHolder>().Select(h=>h.CardModel).ToArray();
            Verify(visible.SequenceEqual(new CardModel[]{chosen,a,b}),"native selection return then visible leftmost order");
            source.InvokeExecutionFinished();Verify(handContainer.GetChildCount()==3,"return callback is one shot");
        }
        finally{patch.UnpatchAll(patch.Id);testHand=null;view.Free();LocalContext.NetId=null;}
        GD.Print("OTTO_CONFIRMED_NATIVE_HAND_RETURN_OK");
    }
