// 一次性验收：只在内存中建立对局，不启动地图、不保存用户的游戏进度。
[ModInitializer(nameof(Initialize))]
public static class Probe
{
    private static int count;
    public static void Initialize() => Callable.From(Check).CallDeferred();
    private static void Verify(bool condition, string message)
    {
        if(!condition) throw new Exception(message);
        count++;
    }
    private static void SetInProgress(bool value) =>
        typeof(CombatManager).GetProperty("IsInProgress")!.SetValue(CombatManager.Instance,value);

    private static async void Check()
    {
        var tree=(SceneTree)Engine.GetMainLoop();
        try
        {
            for(int i=0;i<40;i++) await tree.ToSignal(tree,SceneTree.SignalName.ProcessFrame);
            var canonical=ModelDb.Relic<OttoTalisman>();
            Verify(canonical.Title.GetFormattedText()=="棍亩","遗物中文名称");
            Verify(canonical.DynamicDescription.GetFormattedText().Contains("1"),"遗物动态效果文本");
            Verify(canonical.Rarity==RelicRarity.Starter,"初始稀有度");
            Verify(canonical.Icon!=null && canonical.Icon.GetWidth()==92,"原图小图标");
            Verify(canonical.BigIcon!=null && canonical.IconOutline!=null,"大图及轮廓资源");
            Verify(canonical.HoverTipsExcludingRelic.Any(),"包含原版无实体说明");
            Verify(canonical.Pool is OttoRelicPool,"遗物注册到OTTO独立池");
            Verify(!ModelDb.RelicPool<IroncladRelicPool>().AllRelics.Any(r=>r is OttoTalisman),"原版遗物池未注入OTTO遗物");
            Verify(ModelDb.Character<Ironclad>().StartingRelics.Single() is BurningBlood,"铁甲战士仍携带燃烧之血");
            foreach(var unlock in new[]{UnlockState.none,UnlockState.all})
            {
                var original=ModelDb.RelicPool<IroncladRelicPool>().GetUnlockedRelics(unlock).Where(r=>r.Rarity!=RelicRarity.Starter).Select(r=>r.Id).ToHashSet();
                var actual=ModelDb.RelicPool<OttoRelicPool>().GetUnlockedRelics(unlock).Where(r=>r.Rarity!=RelicRarity.Starter).Select(r=>r.Id).ToHashSet();
                Verify(original.SetEquals(actual),"原有普通遗物奖励及解锁保持一致");
            }
            TestMode.IsOn=true;
            var owner=Player.CreateForNewRun<OttoCharacter>(UnlockState.all,111);
            var ally=Player.CreateForNewRun<Ironclad>(UnlockState.all,112);
            Verify(owner.Relics.Count==1 && owner.Relics[0] is OttoTalisman,"OTTO新对局只携带棍亩");
            Verify(owner.Deck.Cards.Count==11,"初始牌组保持11张");
            var relic=owner.GetRelic<OttoTalisman>()!;
            Verify(RelicModel.FromSerializable(relic.ToSerializable()) is OttoTalisman,"遗物存档ID可还原");
            var run=RunState.CreateForTest([owner,ally],seed:"OTTO-RELIC-CHECK");
            var state=new CombatState(runState:run);
            state.AddPlayer(owner);state.AddPlayer(ally);owner.ResetCombatState();ally.ResetCombatState();
            var enemy=state.CreateCreature(ModelDb.Monster<MockAttackMonster>().ToMutable(),CombatSide.Enemy,"enemy");state.AddCreature(enemy);
            typeof(CombatManager).GetField("_state",BindingFlags.NonPublic|BindingFlags.Instance)!.SetValue(CombatManager.Instance,state);
            var room=new CombatRoom(state);
            // 复现入战斗房间、尚未开始首回合的真实阶段，确认无实体可以在此时施加。
            SetInProgress(false);
            await relic.AfterRoomEntered(null!);
            Verify(!owner.Creature.HasPower<IntangiblePower>(),"非战斗房间不触发");
            await Hook.AfterRoomEntered(run,room);
            Verify(owner.Creature.GetPowerAmount<IntangiblePower>()==1,"开战自动获得1层无实体");
            Verify(!ally.Creature.HasPower<IntangiblePower>(),"不把无实体给予其他玩家");
            SetInProgress(true);
            var ctx=new ThrowingPlayerChoiceContext();
            int hp=owner.Creature.CurrentHp;
            await CreatureCmd.Damage(ctx,owner.Creature,20,ValueProp.Unpowered,enemy);
            Verify(owner.Creature.CurrentHp==hp-1,"20点伤害被原版无实体减为1");
            await owner.Creature.GetPower<IntangiblePower>()!.AfterSideTurnEnd(ctx,CombatSide.Player,[owner.Creature]);
            Verify(owner.Creature.GetPowerAmount<IntangiblePower>()==1,"玩家回合末不提前消失");
            await owner.Creature.GetPower<IntangiblePower>()!.AfterSideTurnEnd(ctx,CombatSide.Enemy,[enemy]);
            Verify(!owner.Creature.HasPower<IntangiblePower>(),"敌方回合末正常失去1层");
            await relic.AfterSideTurnStart(CombatSide.Player,[owner.Creature],state);
            Verify(!owner.Creature.HasPower<IntangiblePower>(),"后续回合不会重复获得");
            hp=owner.Creature.CurrentHp;
            await CreatureCmd.Damage(ctx,owner.Creature,8,ValueProp.Unpowered,enemy);
            Verify(owner.Creature.CurrentHp==hp-8,"状态到期后恢复正常受伤");
            SetInProgress(false);
            await Hook.AfterRoomEntered(run,new CombatRoom(state));
            Verify(owner.Creature.GetPowerAmount<IntangiblePower>()==1,"下一场战斗再次获得1层");
            GD.Print($"OTTO_RELIC_OK: {count} assertions passed; starting relic, icons, pools, Intangible and next combat");
        }
        catch(Exception error){GD.PrintErr("OTTO_RELIC_FAILED: "+error);}
        finally {TestMode.IsOn=false;tree.Quit();}
    }
}
